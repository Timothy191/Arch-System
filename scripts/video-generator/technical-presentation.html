-- Enforce Department & Role Isolation on Shift Logs
ALTER TABLE public.shift_reports ENABLE ROW LEVEL SECURITY;

CREATE POLICY "shift_reports_department_isolation"
ON public.shift_reports FOR ALL
USING (
  auth.uid() IN (
    SELECT user_id FROM public.employees
    WHERE department = shift_reports.department
    OR role IN ('GENERAL_MANAGER', 'ADMIN')
  )
);

Decoupling heavy SLA calculations and SCADA telemetry archiving from the primary UI server using edge workers.

export class ShiftReportWorkflow extends WorkflowEntrypoint {
  async run(event: WorkflowEvent, step: WorkflowStep) {
    // Step 1: Validate SLA against SCADA limits
    const report = await step.do("validate-sla", async () => {
      return evaluateShiftCompliance(event.payload);
    });

    // Step 2: Durable Cloudflare R2 Storage
    await step.do("archive-telemetry", async () => {
      return env.SPATIAL_TELEMETRY_BUCKET.put(report.key, report.data);
    });
  }
}
